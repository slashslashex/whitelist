<?php
// Реальная ссылка на подписку
$real_url = "https://raw.githubusercontent.com/EtoNeYaProject/etoneyaproject.github.io/refs/heads/main/2";

// Скачиваем содержимое
$content = @file_get_contents($real_url);

if ($content === false) {
    http_response_code(503);
    die("Service unavailable: cannot fetch subscription");
}

// Проверяем, не пришла ли ошибка 404 и т.д.
if (strpos($http_response_header[0] ?? '', '404') !== false) {
    http_response_code(404);
    die("Subscription not found");
}

// Определяем тип контента:
// Если строка выглядит как base64 (только латиница, цифры, +, /, =) — отдаем как subscription
if (preg_match('/^[A-Za-z0-9+\/=]+$/', trim($content))) {
    header('Content-Type: application/vnd.v2ray.subscription-singular');
} else {
    header('Content-Type: text/plain; charset=utf-8');
}

// Разрешаем кеширование на 15 минут (опционально)
header('Cache-Control: max-age=900');

echo $content;
?>
